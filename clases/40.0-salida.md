The user wants a self-contained HTML page that displays a real-time analog clock. I need to create a complete HTML file with CSS and JavaScript that draws an analog clock face with hour, minute, and second hands that update in real-time. I'll use a canvas element or CSS to create the clock.

Let me go with a canvas-based approach for a clean, smooth analog clock.
</think>

```html
<!DOCTYPE html>
<html lang="es">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Reloj Analógico</title>
    <style>
        * { margin: 0; padding: 0; box-sizing: border-box; }
        body {
            display: flex;
            justify-content: center;
            align-items: center;
            min-height: 100vh;
            background: #1a1a2e;
            font-family: 'Segoe UI', sans-serif;
        }
        .clock-container {
            position: relative;
            display: flex;
            flex-direction: column;
            align-items: center;
            gap: 20px;
        }
        canvas {
            border-radius: 50%;
            box-shadow:
                0 0 30px rgba(0,0,0,0.5),
                0 0 60px rgba(100, 100, 255, 0.1),
                inset 0 0 30px rgba(0,0,0,0.1);
        }
        .digital {
            color: #e0e0e0;
            font-size: 1.4rem;
            letter-spacing: 4px;
            font-variant-numeric: tabular-nums;
            text-shadow: 0 0 10px rgba(100, 200, 255, 0.3);
        }
    </style>
</head>
<body>
    <div class="clock-container">
        <canvas id="clock" width="400" height="400"></canvas>
        <div class="digital" id="digital"></div>
    </div>

    <script>
        const canvas = document.getElementById('clock');
        const ctx = canvas.getContext('2d');
        const digital = document.getElementById('digital');
        const R = canvas.width / 2;
        const RADIUS = R - 20;

        function drawClock() {
            const now = new Date();
            const h = now.getHours() % 12;
            const m = now.getMinutes();
            const s = now.getSeconds();
            const ms = now.getMilliseconds();

            const smoothS = s + ms / 1000;
            const smoothM = m + smoothS / 60;
            const smoothH = h + smoothM / 60;

            ctx.clearRect(0, 0, canvas.width, canvas.height);

            // Fondo
            const bgGrad = ctx.createRadialGradient(R, R, 0, R, R, RADIUS + 10);
            bgGrad.addColorStop(0, '#2a2a4a');
            bgGrad.addColorStop(0.85, '#1e1e3a');
            bgGrad.addColorStop(1, '#12122a');
            ctx.beginPath();
            ctx.arc(R, R, RADIUS + 10, 0, Math.PI * 2);
            ctx.fillStyle = bgGrad;
            ctx.fill();

            // Borde
            ctx.beginPath();
            ctx.arc(R, R, RADIUS + 10, 0, Math.PI * 2);
            ctx.strokeStyle = '#3a3a6a';
            ctx.lineWidth = 3;
            ctx.stroke();

            // Marcas de minutos
            for (let i = 0; i < 60; i++) {
                const angle = (i * 6 - 90) * Math.PI / 180;
                const isHour = i % 5 === 0;
                const outerR = RADIUS - 5;
                const innerR = isHour ? RADIUS - 28 : RADIUS - 15;

                ctx.beginPath();
                ctx.moveTo(R + Math.cos(angle) * innerR, R + Math.sin(angle) * innerR);
                ctx.lineTo(R + Math.cos(angle) * outerR, R + Math.sin(angle) * outerR);

                if (isHour) {
                    ctx.strokeStyle = '#c0c0e0';
                    ctx.lineWidth = 3;
                } else {
                    ctx.strokeStyle = '#505080';
                    ctx.lineWidth = 1;
                }
                ctx.lineCap = 'round';
                ctx.stroke();
            }

            // Números
            ctx.fillStyle = '#d0d0f0';
            ctx.font = 'bold 22px "Segoe UI", sans-serif';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            for (let i = 1; i <= 12; i++) {
                const angle = (i * 30 - 90) * Math.PI / 180;
                const numR = RADIUS - 48;
                const x = R + Math.cos(angle) * numR;
                const y = R + Math.sin(angle) * numR;
                ctx.fillText(i, x, y);
            }

            // Agujas
            function drawHand(hourFraction, length, width, color, shadowColor) {
                const angle = (hourFraction * 30 - 90) * Math.PI / 180;
                const endX = R + Math.cos(angle) * length;
                const endY = R + Math.sin(angle) * length;
                const tailX = R - Math.cos(angle) * 20;
                const tailY = R - Math.sin(angle) * 20;

                ctx.save();
                ctx.shadowColor = shadowColor || 'transparent';
                ctx.shadowBlur = 8;
                ctx.beginPath();
                ctx.moveTo(tailX, tailY);
                ctx.lineTo(endX, endY);
                ctx.strokeStyle = color;
                ctx.lineWidth = width;
                ctx.lineCap = 'round';
                ctx.stroke();
                ctx.restore();
            }

            drawHand(smoothH, RADIUS * 0.5, 6, '#e0e0f0', 'rgba(100,100,200,0.3)');
            drawHand(smoothM, RADIUS * 0.72, 4, '#c0c0e0', 'rgba(100,100,200,0.3)');
            drawHand(smoothS, RADIUS * 0.82, 2, '#ff4060', 'rgba(255,60,80,0.5)');

            // Centro
            ctx.beginPath();
            ctx.arc(R, R, 7, 0, Math.PI * 2);
            const ctrGrad = ctx.createRadialGradient(R - 2, R - 2, 0, R, R, 7);
            ctrGrad.addColorStop(0, '#ff6080');
            ctrGrad.addColorStop(1, '#cc2040');
            ctx.fillStyle = ctrGrad;
            ctx.fill();

            ctx.beginPath();
            ctx.arc(R, R, 3, 0, Math.PI * 2);
            ctx.fillStyle = '#fff';
            ctx.fill();

            // Digital
            const hh = String(now.getHours()).padStart(2, '0');
            const mm = String(m).padStart(2, '0');
            const ss = String(s).padStart(2, '0');
            digital.textContent = `${hh} : ${mm} : ${ss}`;

            requestAnimationFrame(drawClock);
        }

        drawClock();
    </script>
</body>
</html>
```
